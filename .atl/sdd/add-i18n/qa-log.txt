/**
 * Provider del idioma: única fuente de verdad (RF-15).
 * 1. Idioma inicial: localStorage `portfolio-lang` (try/catch) →
 *    detección `navigator.language` /^en/ → "es" (RF-2/RF-3, ver context.ts).
 * 2. UN efecto `[lang]` hace persistencia + sync de `<html lang>` (RF-4),
 *    de modo que también el primer render sincroniza sin código extra.
 * 3. `value` memoizado por idioma: `t`/`pick` solo cambian al cambiar lang.
 */
import { useEffect, useMemo, useState, type ReactNode } from "react";
import { I18nContext, LANG_STORAGE_KEY, readInitialLang } from "./context";
import { readLocalized } from "./localize";
import { en } from "./en";
import { es } from "./es";
import type { I18nContextValue, Lang } from "./types";

/** Wrapper del árbol (única fuente de verdad del idioma, RF-15). */
export function I18nProvider({ children }: { children: ReactNode }) {
  // Initializer sin invocar: corre una vez, puro, compatible con StrictMode.
  const [lang, setLangState] = useState<Lang>(readInitialLang);

  // Persistencia + <html lang>: UN efecto, UNA fuente de verdad (RF-4, RF-15).
  useEffect(() => {
    try {
      window.localStorage.setItem(LANG_STORAGE_KEY, lang);
    } catch {
      // storage bloqueado: falla en silencio, sin errores en consola (CA-6)
    }
    document.documentElement.lang = lang;
  }, [lang]);

  const value = useMemo<I18nContextValue>(() => {
    const setLang = (next: Lang) => setLangState(next); // la escritura la hace el efecto
    return {
      lang,
      setLang,
      toggle: () => setLang(lang === "es" ? "en" : "es"),
      t: lang === "es" ? es : en,
      pick: (field) => readLocalized(field, lang),
    };
  }, [lang]);

  return <I18nContext.Provider value={value}>{children}</I18nContext.Provider>;
}

/**
 * Context + persistencia/detección del idioma, en un fichero sin componentes.
 * Separado de I18nProvider.tsx porque react-refresh/only-export-components
 * exige que un fichero con componentes solo exporte componentes (el design
 * §1.3 agrupaba esto en el Provider; la firma y semántica no cambian).
 */
import { createContext } from "react";
import type { I18nContextValue, Lang } from "./types";

export const I18nContext = createContext<I18nContextValue | null>(null);

/** Clave de persistencia (plan, decisión 2; NO se traduce). */
export const LANG_STORAGE_KEY = "portfolio-lang";

/**
 * Lee el idioma inicial de forma pura:
 * localStorage → navigator.language /^en/ → "es" (fallback, RF-2).
 * Toda lectura de storage va envuelta en try/catch (modo privado /
 * storage bloqueado solo cuesta la detección, nunca un render roto, RNF-4).
 */
export function readInitialLang(): Lang {
  try {
    const stored = window.localStorage.getItem(LANG_STORAGE_KEY);
    if (stored === "en" || stored === "es") return stored; // preferencia guardada GANA (RF-3)
  } catch {
    // storage bloqueado → sigue a detección
  }
  // /^en/ cubre en-US, en-GB, en… (CA-3). Resto de navegadores → "es".
  if (navigator.language?.startsWith("en")) return "en";
  return "es";
}

/**
 * Selección de lado de un campo Localized. Lógica pura, sin React —

 * `pick` del provider es exactamente esta función con el lang del context:

import type { Lang, Localized } from "./types";

/**
 * Devuelve el lado activo del campo; si va VACÍO (string "" o array []),
 * degrada al otro idioma (RNF-4): nunca renderiza undefined/basura.
 * `value.length === 0` sirve para string y para array a la vez.
 */
export function readLocalized<T extends string | readonly string[]>(
  field: Localized<T>,
  lang: Lang,
): T {
  const active = field[lang];
  if (active.length > 0) return active;
  return field[lang === "es" ? "en" : "es"] as T;
}
